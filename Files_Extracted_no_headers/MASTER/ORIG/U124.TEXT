# MAKEFILE -- the LSI-11 assembler, from U124.1_LSI-11_ASSEM_SOURCE
# on #10:: run by MASTER:/ORIG/MAKEFILE after it mounts that disk (MAKE -f
# MASTER:/ORIG/U124 alone: mount it first).  Each program is compiled by PC (PASCALC:)
# when its source or a file it includes is newer than its code file in MASTER:/ORIG/CODE.
S = #10:
C = MASTER:/ORIG/CODE/

all : $(C)ASM11.CODE

$(C)ASM11.CODE : $(S)ASM11.TEXT $(S)ASM1.TEXT $(S)ASM2.TEXT $(S)ASM3.TEXT \
        $(S)ASM4.TEXT $(S)ASM5.TEXT $(S)ASM6.TEXT
        CD $(S)
        PC ASM11 $(C)ASM11.CODE.

.PHONY : all
