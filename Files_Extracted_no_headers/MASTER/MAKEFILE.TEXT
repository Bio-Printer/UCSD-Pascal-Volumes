# MAKEFILE -- everything: what is new and the original UCSD Pascal II.0 programs, each
# part the MAKE of its own makefile, its volume mounted first (MOUNT: emulator 2.14).
# From the shell, with BIGGY on #4, UCSD-C on #5, MASTER on #9, TOOLS on #11 and the
# other images in the same folder on the PC: MAKE -f MASTER:MAKEFILE, or ... BASIC for
# one part (UCSDC DEMOS TESTS SYSC PASCALC BASIC BASIC2 PASM ORIG).  MASTER:README.TEXT.
all : UCSDC DEMOS TESTS SYSC PASCALC BASIC BASIC2 PASM ORIG

# UCSD-C: the compiler, its library, the tools
UCSDC :
        $(MAKE) -f UCSD-C:/SRC/CC/MAKEFILE
        $(MAKE) -f UCSD-C:/SRC/LIB/MAKEFILE
        $(MAKE) -f TOOLS:/SRC/MAKEFILE
DEMOS :
        MOUNT #12 TCEXTRA
        $(MAKE) -f TCEXTRA:MAKEFILE
TESTS :
        MOUNT #12 TCTESTS
        $(MAKE) -f TCTESTS:MAKEFILE
# the operating system in C
SYSC :
        MOUNT #12 SYS-C
        $(MAKE) -f SYS-C:MAKEFILE
# the Pascal compiler in C
PASCALC :
        MOUNT #12 PASCAL-C
        $(MAKE) -f PASCALC:/SRC/MAKEFILE
# BC and B2, the BASIC compilers
BASIC :
        MOUNT #13 BASIC
        $(MAKE) -f BASIC:MAKEFILE
BASIC2 :
        MOUNT #13 BASIC2
        $(MAKE) -f BASIC2:MAKEFILE
# PASM, the P-code assembler, rebuilt by itself (U016.1)
PASM :
        MOUNT #12 U016.1_PCODE_ASSEM
        $(MAKE) -f U016.1:MAKEFILE
# the original programs (PC on PASCALC:)
ORIG :
        MOUNT #12 PASCAL-C
        $(MAKE) -f MASTER:/ORIG/MAKEFILE

.PHONY : all UCSDC DEMOS TESTS SYSC PASCALC BASIC BASIC2 PASM ORIG
