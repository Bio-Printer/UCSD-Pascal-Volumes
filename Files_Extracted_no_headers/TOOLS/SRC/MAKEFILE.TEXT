# MAKEFILE -- every tool on TOOLS:: from the shell, MAKE -f TOOLS:/SRC/MAKEFILE
# (UCSD-C: on line), or MAKE -f TOOLS:/SRC/MAKEFILE TOOLS:VI.CODE for one.  Compiles
# each module whose source or headers changed (-Z; its NAME.OBJ into TOOLS:/OBJ)
# and links the tools whose objects changed onto TOOLS:.  TOOLS:/OBJ is empty on a
# new volume: the first MAKE builds every tool.  MAKE -B ... builds them all again.
S = TOOLS:/SRC/
O = TOOLS:/OBJ/
I = UCSD-C:/INCLUDE/
ZC = $(CC) -O $(O) -Z -C

all : TOOLS:VI.CODE TOOLS:GREP.CODE TOOLS:CAT.CODE TOOLS:HEXDUMP.CODE \
        TOOLS:OBJCOPY.CODE TOOLS:MORE.CODE TOOLS:LESS.CODE TOOLS:SED.CODE \
        TOOLS:DIFF.CODE TOOLS:WC.CODE TOOLS:LC.CODE TOOLS:KRUNCH.CODE \
        TOOLS:GROW.CODE TOOLS:SHRINK.CODE TOOLS:PATCH.CODE TOOLS:MAN.CODE \
        TOOLS:BINDERC.CODE TOOLS:ECHO.CODE TOOLS:HEAD.CODE TOOLS:TAIL.CODE \
        TOOLS:SORT.CODE TOOLS:UNIQ.CODE TOOLS:TR.CODE TOOLS:CUT.CODE TOOLS:TEE.CODE \
        TOOLS:MAKE.CODE TOOLS:TOUCH.CODE TOOLS:DATE.CODE TOOLS:MOUNT.CODE
