; PASMGEN -- PASM's instructions, directives, procedures and segments

        .GLOBAL OPC, OPF, ENDING

; ---- the procedure's code

; OB(B): the byte B at the end of the procedure's code
        .PROC   OB
        .PARAM  B
        LDO     CLEN
        LIT     MAXCODE
        LESI
        FJP     FULL
        LDL     B
        CGP     PUTB
        RNP     0
FULL:   LSA     'procedure too long'
        CGP     ERRS
        RNP     0
        .END

; PUTB(B): B at the end of CODE (its tables too: no limit but CODE's size)
        .PROC   PUTB
        .PARAM  B
        LAO     CODE
        LDO     CLEN
        LDL     B
        STB
        LDO     CLEN
        SLDC    1
        ADI
        SRO     CLEN
        RNP     0
        .END

; PUTW(W): the word W at the end of CODE, its low byte first
        .PROC   PUTW
        .PARAM  W
        LDL     W
        LDCI    255
        LAND
        CGP     PUTB
        LDL     W
        SLDC    0
        CGP     HIBYTE
        CGP     PUTB
        RNP     0
        .END

; BIG(V, F): V as a big operand: one byte below 128 (unless F), else two
        .PROC   BIG
        .PARAM  V, F
        LDL     V
        SLDC    0
        LESI
        FJP     OK
        LSA     'out of range (0..32767)'
        CGP     ERRS
OK:     LDL     V
        LIT     128
        LESI
        LDL     F
        LNOT
        LAND
        FJP     TWO
        LDL     V
        CGP     OB
        RNP     0
TWO:    LDL     V
        LDCI    256
        DVI
        LIT     128
        LOR
        CGP     OB
        LDL     V
        LDCI    255
        LAND
        CGP     OB
        RNP     0
        .END

; BYTE(V): V as a byte operand (0..255)
        .PROC   BYTE
        .PARAM  V
        LDL     V
        SLDC    0
        LESI
        LDL     V
        LDCI    255
        GRTI
        LOR
        FJP     OK
        LSA     'out of range (0..255)'
        CGP     ERRS
OK:     LDL     V
        CGP     OB
        RNP     0
        .END

; WORD(V): V as a word operand, its low byte first
        .PROC   WORD

        LAND
        CGP     OB
        LDL     V
        SLDC    0
        CGP     HIBYTE
        CGP     OB

        .END

        .END

        SLDC    1
        STL     R

        .END
