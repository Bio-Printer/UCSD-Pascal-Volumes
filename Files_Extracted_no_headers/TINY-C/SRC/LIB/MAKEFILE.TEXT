# MAKEFILE -- the C library: from the shell, MAKE -f TINY-C:/SRC/LIB/MAKEFILE
# (MAKE.CODE is on TOOLS:).  Compiles each module whose source or headers changed
# (-Z; its NAME.OBJ into TINY-C:/OBJ), then joins them into TINY-C:/LIB/TCLIB2.OBJ,
# identical to TINY-C:/LIB/TCLIB.OBJ (CMPCODE says so) -- in parts
# (TINY-C:/OBJ/TCLIBPn) where one -J would be longer than the 150 characters a line
# of CC's batch file holds.  MAKE -B ... compiles them all again.
S = TINY-C:/SRC/LIB/
O = TINY-C:/OBJ/
I = TINY-C:/INCLUDE/

TINY-C:/LIB/TCLIB2.OBJ : $(O)TCLIBP1.OBJ $(O)STRTOLD.OBJ $(O)SUBVOL.OBJ \
        $(O)TCRT.OBJ
        $(CC) -O $(O) -J TINY-C:/LIB/TCLIB2=$(O)TCLIBP1,STRTOLD,SUBVOL,TCRT
$(O)TCLIBP1.OBJ : $(O)ASSERT.OBJ $(O)CONIO.OBJ $(O)CTYPE.OBJ $(O)DIRS.OBJ \
        $(O)FILEARGS.OBJ $(O)FLTFMT.OBJ $(O)FTIME.OBJ $(O)IO.OBJ $(O)MATH.OBJ \
        $(O)MEMSCAN.OBJ $(O)PEXEC.OBJ $(O)REDIRECT.OBJ $(O)SETJMP.OBJ $(O)STDIO.OBJ \
        $(O)STDIOBLK.OBJ $(O)STDLIB.OBJ $(O)STRING.OBJ
        $(CC) -O $(O) -J $(O)TCLIBP1=ASSERT,CONIO,CTYPE,DIRS,FILEARGS,FLTFMT,FTIME,IO,MATH,MEMSCAN,PEXEC,REDIRECT,SETJMP,STDIO,STDIOBLK,STDLIB,STRING

$(O)%.OBJ : $(S)%.C
        $(CC) -O $(O) -Z -C $<

# each module's headers: it is compiled again when one of them changes
