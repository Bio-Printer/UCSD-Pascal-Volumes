; LIBS -- rebuild the C library: CC @TINY-C:/SRC/LIB/LIBS from the shell
; (or X(ecute TINY-C:CC and answer @TINY-C:/SRC/LIB/LIBS).  Compiles every library
; module here (-O: each NAME.OBJ into TINY-C:/OBJ), then joins them into
; TINY-C:/LIB/TCLIB2.OBJ (in parts, TINY-C:/OBJ/TCLIBPn, when one line would be
; longer than the 150 characters CC reads).
-O TINY-C:/OBJ
-Z -C ASSERT
-Z -C CONIO
-Z -C CTYPE
-Z -C DIRS
-Z -C FILEARGS
-Z -C FLTFMT
-Z -C FTIME
-Z -C IO
-Z -C MATH
-Z -C MEMSCAN
-Z -C PEXEC
-Z -C REDIRECT
-Z -C SETJMP
-Z -C STDIO
-Z -C STDIOBLK
-Z -C STDLIB
-Z -C STRING
-Z -C STRTOLD
-Z -C SUBVOL
-Z -C TCRT
-J TINY-C:/OBJ/TCLIBP1=ASSERT,CONIO,CTYPE,DIRS,FILEARGS,FLTFMT,FTIME,IO,MATH,MEMSCAN,PEXEC,REDIRECT,SETJMP,STDIO,STDIOBLK,STDLIB,STRING
-J TINY-C:/LIB/TCLIB2=TINY-C:/OBJ/TCLIBP1,STRTOLD,SUBVOL,TCRT
