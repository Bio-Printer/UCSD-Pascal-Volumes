# MAKEFILE -- BC, rebuilt: from the shell, MAKE -f BASIC:/SRC/MAKEFILE (Tiny-C's
# MAKE, on TOOLS:).  Compiles each module whose source, or a file it #INCLUDEs, is newer than
# its object (NAME.OBJ beside it), and links each program whose modules changed, as
# BASIC:BC2.CODE ... (its map BASIC:BC2.MAP: -MAP).  MAKE -B ... all of them again.
# MAKE -f BASIC:/SRC/MAKEFILE INSTALL then puts them in the old ones' places (each MOVE
# deletes the old file and renames the new one; when the new one is not there it says
# so and leaves the old one).
BC = BC
S = BASIC:/SRC/
FLAGS =

all : BASIC:BC2.CODE BASIC:BCLEX2.CODE BASIC:BCGEN2.CODE BASIC:BCASM2.CODE \
        BASIC:BCLIST2.CODE BASIC:BCLINK2.CODE BASIC:BCERR2.CODE BASIC:BCDEBUG2.CODE

BASIC:BC2.CODE : $(S)BC.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BC2=$(S)BC,$(S)BCUTIL
BASIC:BCLEX2.CODE : $(S)BCLEX.OBJ $(S)BCKW.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BCLEX2=$(S)BCLEX,$(S)BCKW,$(S)BCUTIL
BASIC:BCGEN2.CODE : $(S)BCGEN.OBJ $(S)BCSTMT.OBJ $(S)BCDECL.OBJ \
        $(S)BCEXPR.OBJ $(S)BCEXGEN.OBJ $(S)BCEMIT.OBJ $(S)BCTOK.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BCGEN2=$(S)BCGEN,$(S)BCSTMT,$(S)BCDECL,$(S)BCEXPR,$(S)BCEXGEN,$(S)BCEMIT,$(S)BCTOK,$(S)BCUTIL
BASIC:BCASM2.CODE : $(S)BCASM.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BCASM2=$(S)BCASM,$(S)BCUTIL
BASIC:BCLIST2.CODE : $(S)BCLIST.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BCLIST2=$(S)BCLIST,$(S)BCUTIL
BASIC:BCLINK2.CODE : $(S)BCLINK.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BCLINK2=$(S)BCLINK,$(S)BCUTIL
BASIC:BCERR2.CODE : $(S)BCERR.OBJ $(S)BCKW.OBJ $(S)BCUTIL.OBJ
        $(BC) -MAP -L BASIC:BCERR2=$(S)BCERR,$(S)BCKW,$(S)BCUTIL
BASIC:BCDEBUG2.CODE : $(S)BCDEBUG.OBJ
        $(BC) -MAP -L BASIC:BCDEBUG2=$(S)BCDEBUG

$(S)%.OBJ : $(S)%.BAS
        $(BC) $(FLAGS) -C $(S)$*

# each module's #INCLUDEs: it is compiled again when one of them changes
$(S)BC.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCUTIL.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCLEX.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCKW.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCGEN.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS $(S)BCGENCOM.BAS
$(S)BCSTMT.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS $(S)BCGENCOM.BAS
$(S)BCDECL.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCEXPR.OBJ : $(S)BCBUILT.BAS $(S)BCCOM.BAS $(S)BCPROCS.BAS \
        $(S)BCGENCOM.BAS
$(S)BCEXGEN.OBJ : $(S)BCBUILT.BAS $(S)BCCOM.BAS $(S)BCPROCS.BAS \
        $(S)BCGENCOM.BAS
$(S)BCEMIT.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS $(S)BCGENCOM.BAS
$(S)BCTOK.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCASM.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS $(S)BCGENCOM.BAS
$(S)BCLIST.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS $(S)BCGENCOM.BAS
$(S)BCLINK.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS
$(S)BCERR.OBJ : $(S)BCCOM.BAS $(S)BCPROCS.BAS

INSTALL :
        MOVE BASIC:BC2.CODE BASIC:BC.CODE
        MOVE BASIC:BC2.MAP BASIC:BC.MAP
        MOVE BASIC:BCLEX2.CODE BASIC:BCLEX.CODE
        MOVE BASIC:BCLEX2.MAP BASIC:BCLEX.MAP
        MOVE BASIC:BCGEN2.CODE BASIC:BCGEN.CODE
        MOVE BASIC:BCGEN2.MAP BASIC:BCGEN.MAP
        MOVE BASIC:BCASM2.CODE BASIC:BCASM.CODE
        MOVE BASIC:BCASM2.MAP BASIC:BCASM.MAP
        MOVE BASIC:BCLIST2.CODE BASIC:BCLIST.CODE
        MOVE BASIC:BCLIST2.MAP BASIC:BCLIST.MAP
        MOVE BASIC:BCLINK2.CODE BASIC:BCLINK.CODE
        MOVE BASIC:BCLINK2.MAP BASIC:BCLINK.MAP
        MOVE BASIC:BCERR2.CODE BASIC:BCERR.CODE
        MOVE BASIC:BCERR2.MAP BASIC:BCERR.MAP
        MOVE BASIC:BCDEBUG2.CODE BASIC:BCDEBUG.CODE
        MOVE BASIC:BCDEBUG2.MAP BASIC:BCDEBUG.MAP

.PHONY : all INSTALL
